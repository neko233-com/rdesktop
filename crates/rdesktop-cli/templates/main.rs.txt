#![cfg_attr(not(debug_assertions), windows_subsystem = "windows")]
use rdesktop_core::{config::AppConfig, renderer::Renderer};
use rdesktop_core::ipc::{FnIpcHandler, IpcMessage, IpcResponse};
#[cfg(feature = "webview")]
use rdesktop_webview::WebViewRenderer as AppRenderer;
#[cfg(all(feature = "chrome", not(feature = "webview")))]
use rdesktop_cef::CefRenderer as AppRenderer;

fn main() -> anyhow::Result<()> {
    let mut config = AppConfig::default();
    config.name = env!("CARGO_PKG_NAME").into();
    config.window.title = config.name.clone();
    let mut renderer = AppRenderer::new(&config)?;
    renderer.init()?;
    renderer.set_ipc_handler(Box::new(FnIpcHandler::new(|msg: IpcMessage| {
        IpcResponse {
            id: msg.id,
            success: msg.cmd == "greet",
            data: if msg.cmd == "greet" {
                serde_json::json!({"message": format!("Hello, {}!", msg.payload["name"].as_str().unwrap_or("World"))})
            } else { serde_json::json!({"error": "Unknown command"}) },
        }
    })));
    #[cfg(feature = "webview")]
    renderer.enable_native_agent_from_env()?;
    let window = renderer.create_window(&config.window)?;
    #[cfg(debug_assertions)]
    if let Ok(url) = std::env::var("RDESKTOP_DEV_URL") {
        renderer.load_url(window, &url)?;
        return Box::new(renderer).run().map_err(Into::into);
    }
    renderer.load_html(window, include_str!("../frontend/index.html"))?;
    Box::new(renderer).run().map_err(Into::into)
}
